## 2. 수정 명세 (`shinseon_cockpit.pyw`)

```python
# Before (Error: no running event loop)
def main():
    app = QApplication(sys.argv)
    loop = QEventLoop(app)
    asyncio.set_event_loop(loop)

    cockpit = ShinseonCockpit()
    cockpit.show()

    # ❌ 루프 실행 전에 create_task 호출하여 충돌 발생!
    asyncio.create_task(cockpit.connect_websocket())

    with loop:
        loop.run_forever()

# After (Clean & Safe QTimer Delay)
def main():
    app = QApplication(sys.argv)
    loop = QEventLoop(app)
    asyncio.set_event_loop(loop)

    cockpit = ShinseonCockpit()
    cockpit.show()

    # 🟢 루프 가동 직후 안전하게 비동기 웹소켓 태스크 스케줄링!
    QTimer.singleShot(50, lambda: asyncio.create_task(cockpit.connect_websocket()))

    with loop:
        loop.run_forever()
```
