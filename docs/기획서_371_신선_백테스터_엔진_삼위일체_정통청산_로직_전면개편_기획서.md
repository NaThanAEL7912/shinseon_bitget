## 2. 백테스터 엔진 Clean Core 개편 상세 명세

### 2.1. 0단계 & 삼위일체 3대 AND 신호 판정
```python
# [0단계]: 필수 듀얼 임계치 검사
stage0_passed = (liq_total >= t_liq) and (abs(oi_speed) >= t_oi)
all_zero = (price_delta_5s == 0.0) and (price_delta_1m == 0.0) and (price_slope_1m == 0.0)

sig_dir = None
if stage0_passed and not all_zero:
    # 🟢 찐 롱: 5s >= 0 AND 1m >= 0 AND Slope >= 0
    if (price_delta_5s >= 0.0) and (price_delta_1m >= 0.0) and (price_slope_1m >= 0.0):
        sig_dir = "LONG"
    # 🔴 찐 숏: 5s <= 0 AND 1m <= 0 AND Slope <= 0
    elif (price_delta_5s <= 0.0) and (price_delta_1m <= 0.0) and (price_slope_1m <= 0.0):
        sig_dir = "SHORT"
    else:
        sig_dir = None
else:
    sig_dir = None
```

### 2.2. 포지션 관리 및 정통 청산(Close to Flat) & 쿨타임 루프
```python
if not is_in:
    # 무포지션 상태: 설정된 쿨타임(cooldown) 경과 후 새로운 유효 신호 발생 시 진입
    if cts < cooldown:
        continue
    if sig_dir in ["LONG", "SHORT"]:
        is_in = True
        direction = sig_dir
        entry_price = cp
        entry_time = cts
else:
    # 포지션 보유 중:
    # 1. 🚨 반대 신호(Opposite Signal) 발생 시 ➔ 60초 락 없이 즉시 100% 전량 청산!
    if sig_dir and (sig_dir != direction):
        closed = True
        reason = f"반대신호청산 ({sig_dir})"
        # 수익 청산 시 tp_cooldown, 손실 청산 시 sl_cooldown 설정값 준수
        cooldown = cts + (tp_cooldown if pnl_cur > 0 else sl_cooldown)

    # 2. 거래소 지정가 손절선(SL) 도달 시 청산
    elif pnl_cur <= -sl_ratio:
        closed = True
        reason = f"손절 (-{sl_pct:.2f}%)"
        cooldown = cts + sl_cooldown

    # 3. 거래소 지정가 익절선(TP) 도달 시 청산
    elif tp_ratio > 0 and pnl_cur >= tp_ratio:
        closed = True
        reason = f"익절 (+{tp_pct:.2f}%)"
        cooldown = cts + tp_cooldown
        
    else:
        # 동일 신호 또는 NONE(관망) 발생 시 ➔ 100% 홀딩 유지!
        pass
```
